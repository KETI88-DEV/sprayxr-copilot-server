# KETI API v1.5 Seam Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `/api/ext/v1/*` on our demo API so the separately-owned KETI `/api/v1` service (`keti-xr-lab.iptime.org:58000`) can drive our AI pipeline by setting one base URL.

**Architecture:** A translation boundary plus three latency tiers. This plan covers **Tier 0 only** (rule/MCP path, sub-second) through to a reference proxy that KETI can run. Every KETI identifier is quarantined in `keti_vocab.py` and `keti_router.py`; nothing downstream sees `R-UP-01` or `W-017`. The existing `/api/sim/*` surface is untouched — both surfaces become adapters over one extended `SimSessionStore`.

**Tech Stack:** Python 3.11.9, FastAPI 0.135.1, Pydantic 2.12.5, pytest, httpx 0.28.1. No new dependencies.

**Source spec:** `docs/superpowers/specs/2026-08-03-keti-api-v1.5-integration-design.md`

## Scope

Covers §9.4 steps 1–4 of the spec — the first deliverable to KETI. **Out of scope:** Tier 1 `copilot/query` (blocked on latency Task 8 setting `query_timeout_ms`), Tier 2 `session/report` (blocked on OQ-3, whether `result` is one JSON string or eight form fields), and Qwen3-VL.

```python
"""Translation boundary for KETI AI Co-Pilot API v1.5 vocabulary.

KETI identifiers must not exist outside this module and the ext router.
Everything downstream keeps the naming fixed by AGENTS.md (TR-L01, A-1..A-3).
"""

from __future__ import annotations

import json
from collections.abc import Mapping
from dataclasses import dataclass
from pathlib import Path
from typing import Literal, cast

CoatingStatus = Literal["normal", "over-app", "under-app"]


@dataclass(frozen=True, slots=True)
class KetiVocab:
    zones: Mapping[str, str]
    scenarios: Mapping[str, str]

    def zone_to_ours(self, keti_zone_id: str) -> str | None:
        """None means unmapped; the caller records it as unresolved and does not move."""
        return self.zones.get(keti_zone_id)

    def zone_to_keti(self, our_zone_id: str) -> str:
        """Identity fallback on purpose: recommended_flow must never come back empty."""
        for keti_id, ours in self.zones.items():
            if ours == our_zone_id:
                return keti_id
        return our_zone_id

    def scenario_to_level(self, scenario_id: str) -> str | None:
        return self.scenarios.get(scenario_id)


def load_vocab(path: Path) -> KetiVocab:
    raw = cast(dict[str, object], json.loads(path.read_text(encoding="utf-8")))
    zones = cast(dict[str, str], raw.get("zones", {}))
    scenarios = cast(dict[str, str], raw.get("scenarios", {}))
    return KetiVocab(zones=dict(zones), scenarios=dict(scenarios))
```

```python
"""Wire models for KETI AI Co-Pilot API v1.5.

Field names mirror the KETI document exactly so their handler stays a passthrough.
Translation into our vocabulary happens in keti_vocab, never here.
"""

from __future__ import annotations

from typing import ClassVar, Literal

from pydantic import BaseModel, ConfigDict

from trainer.demo_api.keti_vocab import CoatingStatus


class KetiModel(BaseModel):
    model_config: ClassVar[ConfigDict] = ConfigDict(extra="forbid")


class SessionStartRequest(KetiModel):
    session_id: str
    user_id: str
    scenario_id: str
    started_at: str


class SessionEndRequest(KetiModel):
    session_id: str
    ended_at: str


class SequenceSetting(KetiModel):
    tool_type: str
    pressure: float
    nozzle_size: str


class ToolPosRot(KetiModel):
    pos: list[float]
    rot: list[float]


class JointPose(KetiModel):
    pos: list[float]
    rot: list[float]


class SequenceFrame(KetiModel):
    index: int
    time: str
    tool_pos_rot: ToolPosRot
    # v1.5 types this Array[Object]; every documented sample is a joint-name map (OQ-8).
    skeleton_joints: dict[str, JointPose]


class SequenceResult(KetiModel):
    avg_speed_mms: float
    avg_distance_mm: float
    avg_thickness_um: float
    status: CoatingStatus


class SequenceUpload(KetiModel):
    session_id: str
    user_id: str
    scenario_id: str
    zone_id: str
    sequence_id: str
    sequence_setting: SequenceSetting
    sequence_info: list[SequenceFrame]
    sequence_result: SequenceResult


class AnomalyEvent(KetiModel):
    """Byte-shape of the data: payload on their anomaly_detection SSE event."""

    session_id: str
    zone_id: str
    sequence_id: str
    thickness: float
    status: str
    message: str


class NoAnomaly(KetiModel):
    is_anomaly: Literal[False] = False


class ApiResponse(BaseModel):
    status: Literal["SUCCESS", "ERROR"]
    code: int
    message: str
    data: dict[str, object]


def ok(data: dict[str, object]) -> ApiResponse:
    return ApiResponse(
        status="SUCCESS", code=200, message="Operation completed successfully", data=data
    )
```

```python
    resolve the session-id-free copilot query by user. ``current()`` returns the
    most recently updated session, which is exactly the previous single-slot
    behavior whenever one session is live. We are a cache, not a system of
    record - KETI's MySQL is authoritative, so eviction loses nothing durable.
    """

    def __init__(self, max_sessions: int = DEFAULT_MAX_SESSIONS) -> None:
        self._lock = threading.Lock()
        self._sessions: OrderedDict[str, SimSessionState] = OrderedDict()
        self._max_sessions = max_sessions

    def _touch(self, state: SimSessionState) -> None:
        """Caller must hold the lock. Most recent lands at the end."""
        self._sessions[state.session_id] = state
        self._sessions.move_to_end(state.session_id)
        while len(self._sessions) > self._max_sessions:
            _ = self._sessions.popitem(last=False)

    def start(
        self,
        *,
        session_id: str,
        level_id: str,
        recommended_flow: Sequence[str],
        user_id: str | None = None,
    ) -> SimSessionState:
        flow = tuple(recommended_flow)
        if not flow:
            raise ValueError("recommended_flow must not be empty")
        state = SimSessionState(
            session_id=session_id,
            level_id=level_id,
            phase="start",
            recommended_flow=flow,
            current_zone_id=flow[0],
            completed_zone_ids=(),
            last_event_id=None,
            anomaly_count=0,
            unresolved_zone_id=None,
            updated_at=_now(),
            user_id=user_id,
        )
        with self._lock:
            self._touch(state)
        return state

    def runtime(
        self, *, session_id: str, event_id: str, target_zone_id: str | None, anomaly: bool
    ) -> SimSessionState | None:
        with self._lock:
            state = self._sessions.get(session_id)
            if state is None:
                return None

            current_zone_id = state.current_zone_id
            completed = state.completed_zone_ids
            unresolved = state.unresolved_zone_id

            if target_zone_id is not None:
                if target_zone_id in state.recommended_flow:
                    index = state.recommended_flow.index(target_zone_id)
                    current_zone_id = target_zone_id
                    completed = state.recommended_flow[:index]
                    unresolved = None
                else:
                    unresolved = target_zone_id

            updated = replace(
                state,
                phase="runtime",
                current_zone_id=current_zone_id,
                completed_zone_ids=completed,
                last_event_id=event_id,
                anomaly_count=state.anomaly_count + (1 if anomaly else 0),
                unresolved_zone_id=unresolved,
                updated_at=_now(),
            )
            self._touch(updated)
            return updated

    def end(self, *, session_id: str) -> SimSessionState | None:
        with self._lock:
            state = self._sessions.get(session_id)
            if state is None:
                return None
            updated = replace(state, phase="session_end", updated_at=_now())
            self._touch(updated)
            return updated

    def current(self) -> SimSessionState | None:
        with self._lock:
            if not self._sessions:
                return None
            return next(reversed(self._sessions.values()))

    def get(self, session_id: str) -> SimSessionState | None:
        with self._lock:
            return self._sessions.get(session_id)

    def latest_for_user(self, user_id: str) -> SimSessionState | None:
        with self._lock:
            for state in reversed(self._sessions.values()):
                if state.user_id == user_id:
                    return state
        return None
```

```python
"""Tier 0 seam for the KETI /api/v1 service - rule path only, never the LLM.

A spray pass lasts seconds. An LLM verdict would warn about a zone the trainee
already left, so this module must not import AgentPipeline.
"""

from __future__ import annotations

import logging
from typing import Protocol

from fastapi import APIRouter

from trainer.demo_api.keti_contracts import (
    AnomalyEvent,
    ApiResponse,
    NoAnomaly,
    SequenceUpload,
    SessionEndRequest,
    SessionStartRequest,
    ok,
)
from trainer.demo_api.keti_vocab import KetiVocab
from trainer.demo_api.sequence import default_snapshot
from trainer.demo_api.sim_session import SimSessionStore

logger = logging.getLogger(__name__)

DEFAULT_LEVEL_ID = "TR-L01"

_FALLBACK_MESSAGE = {
    "over-app": "도막 두께 과다 도포가 감지되었습니다. 분사량을 줄이고 이동 속도를 높이세요.",
    "under-app": "도막 두께 미달이 감지되었습니다. 이동 속도를 낮추고 이격 거리를 확인하세요.",
}


class VerdictClient(Protocol):
    async def explain(self, upload: SequenceUpload) -> str | None:
        """Return coaching text, or None when the source is unavailable."""
        ...


def create_keti_router(
    store: SimSessionStore, vocab: KetiVocab, verdict_client: VerdictClient
) -> APIRouter:
    router = APIRouter(prefix="/api/ext/v1", tags=["keti"])

    @router.post("/session/open")
    async def session_open(payload: SessionStartRequest) -> ApiResponse:
        level_id = vocab.scenario_to_level(payload.scenario_id) or DEFAULT_LEVEL_ID
        our_flow = [zone.zone_id for zone in default_snapshot(level_id).zones]
        _ = store.start(
            session_id=payload.session_id,
            level_id=level_id,
            recommended_flow=our_flow,
            user_id=payload.user_id,
        )
        return ok(
            {
                "session_id": payload.session_id,
                "recommended_flow": [vocab.zone_to_keti(zone) for zone in our_flow],
                "guidance": "권장 순서에 따라 상단 구역부터 도장하세요.",
            }
        )

    @router.post("/sequence/verdict")
    async def sequence_verdict(payload: SequenceUpload) -> ApiResponse:
        status = payload.sequence_result.status
        is_anomaly = status != "normal"

        # Unity already decided the status (OQ-7). We explain it, we do not reclassify.
        _ = store.runtime(
            session_id=payload.session_id,
            event_id=payload.sequence_id,
            target_zone_id=vocab.zone_to_ours(payload.zone_id) or payload.zone_id,
            anomaly=is_anomaly,
        )

        if not is_anomaly:
            return ok(NoAnomaly().model_dump(mode="json"))

        message: str | None = None
        try:
            message = await verdict_client.explain(payload)
        except Exception:
            logger.exception("verdict explanation failed; falling back to the rule message")

        event = AnomalyEvent(
            session_id=payload.session_id,
            zone_id=payload.zone_id,
            sequence_id=payload.sequence_id,
            thickness=payload.sequence_result.avg_thickness_um,
            status=status,
            message=message or _FALLBACK_MESSAGE[status],
        )
        return ok(event.model_dump(mode="json"))

    @router.post("/session/close")
    async def session_close(payload: SessionEndRequest) -> ApiResponse:
        _ = store.end(session_id=payload.session_id)
        return ok({"session_id": payload.session_id})

    return router
```

Note: `vocab.zone_to_ours(...) or payload.zone_id` passes the raw KETI id through when unmapped. The store's existing rule then records it as `unresolved_zone_id` because it is not in `recommended_flow` — exactly the interim behavior spec §6 requires.

```python
"""Rule-tier explanation source: MCP anomaly_detection over HTTP.

Returns None on any failure. MCP answers errors with HTTP 200, so the status
field decides success - never the status code.
"""

from __future__ import annotations

import logging
from typing import cast

import httpx

from trainer.demo_api.keti_contracts import SequenceUpload

logger = logging.getLogger(__name__)

TIMEOUT_SEC = 2.0


class McpVerdictClient:
    def __init__(self, base_url: str) -> None:
        self._base_url = base_url.rstrip("/")

    async def explain(self, upload: SequenceUpload) -> str | None:
        body = {
            "measurements": [upload.sequence_result.avg_thickness_um],
            "spec": {"nominal_um": 100.0, "tolerance_rule": "90/10"},
            "zone_id": upload.zone_id,
        }
        try:
            async with httpx.AsyncClient(timeout=TIMEOUT_SEC) as client:
                response = await client.post(
                    f"{self._base_url}/anomaly_detection/detect", json=body
                )
            payload = cast(dict[str, object], response.json())
        except Exception:
            logger.warning("MCP anomaly_detection unreachable", exc_info=True)
            return None

        if payload.get("status") != "success":
            return None
        result = cast(dict[str, object], payload.get("result", {}))
        message = result.get("recommendation")
        return message if isinstance(message, str) and message else None
```

```python
Hand this to KETI. It shows every handler as a thin passthrough to our

import uvicorn
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

_ROUTES = {
    "/api/v1/sessions/start": "/api/ext/v1/session/open",
    "/api/v1/sessions/end": "/api/ext/v1/session/close",
    "/api/v1/telemetry/sequence": "/api/ext/v1/sequence/verdict",

def create_proxy_app(ext_base_url: str) -> FastAPI:
    app = FastAPI(title="KETI /api/v1 reference proxy")
    app.state.ext_base_url = ext_base_url.rstrip("/")
    app.state.http_client = None

    async def _forward(request: Request, target: str) -> JSONResponse:
        body = cast(object, await request.json())
        injected = app.state.http_client
        if injected is not None:
            response = injected.post(target, json=body)
        else:
            async with httpx.AsyncClient(timeout=10.0) as client:
                response = await client.post(f"{app.state.ext_base_url}{target}", json=body)
        return JSONResponse(status_code=response.status_code, content=response.json())

    for source, target in _ROUTES.items():
        def _make(bound_target: str):
            async def _handler(request: Request) -> JSONResponse:
                return await _forward(request, bound_target)

            return _handler

        _ = app.post(source)(_make(target))

    return app


def main() -> None:
    parser = argparse.ArgumentParser()
    _ = parser.add_argument("--ext-base-url", default="http://127.0.0.1:8120")
    _ = parser.add_argument("--host", default="0.0.0.0")
    _ = parser.add_argument("--port", default=8140, type=int)
    args = cast(dict[str, object], vars(parser.parse_args()))
    app = create_proxy_app(cast(str, args["ext_base_url"]))
    uvicorn.run(app, host=cast(str, args["host"]), port=cast(int, args["port"]))


if __name__ == "__main__":
    main()
```
